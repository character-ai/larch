def normalize_design_severity(raw):
    value = (raw or "").strip().lower()
    aliases = {
        "major": "important",
        "important": "important",
        "minor": "latent",
        "latent": "latent",
        "nit": "nit",
        "none": "(none)",
        "(none)": "(none)",
